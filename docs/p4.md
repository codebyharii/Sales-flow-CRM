# Phase 4: Auth Modal & All 4 UI States — Execution Blueprint

> **PRD Priority:** MUST (REQ-02, Screen 02 Wireframe — 20% Evaluation Weight)  
> **Phase Goal:** Implement the interactive Auth Modal overlay with Sign Up / Log In toggle tabs, all 4 explicit UI states (`default`, `submitting`, `error`, `success`), reserved-height error container to prevent layout jump, keyboard accessibility (`Esc`), and mobile full-screen responsiveness.

---

## 1. Overview & PRD Screen 02 Requirements

### Key UI/UX Specifications (PRD Section 05 - Screen 02)
1. **Modal Overlay:** Rendered as an overlay directly over the landing page. Never navigates away to a separate route.
2. **Tab Switcher:** Single modal with top tab toggle: **Sign Up** vs **Log In**.
   - Sign Up shows: Name (optional), Email, Password fields.
   - Log In shows: Email, Password fields only.
3. **4 Explicit States:**
   - **`default`:** Opens on Sign Up tab (or Log In tab if prior login flag in `localStorage` or `?login=1` URL parameter).
   - **`submitting`:** Fields locked (`disabled`), submit button disabled showing a loading spinner + label change ("Creating account..." / "Logging in...").
   - **`error`:** Displays inline error message inside a **reserved-height error container** above the submit button (prevents layout jumping/shifting). Form inputs preserved.
   - **`success`:** Closes modal and redirects user straight to `/booking`.
4. **Accessibility & Keyboard Handling:**
   - `Escape` key press closes modal.
   - Backdrop click closes modal.
   - All input fields use semantic `<label>` elements.
5. **Mobile Viewport (≤640px):** Modal becomes a full-screen overlay for touch devices.

## 3. Step-by-Step Implementation Code

### Step 1: Create Auth Modal Component (`src/components/AuthModal.tsx`)

```tsx
"use client";

import React, { useState, useEffect, useRef } from "react";
import { useRouter } from "next/navigation";
import { signIn } from "next-auth/react";
import { X, Loader2, AlertCircle, Mail, Lock, User } from "lucide-react";

interface AuthModalProps {
  isOpen: boolean;
  onClose: () => void;
  initialTab?: "signup" | "login";
}

export function AuthModal({ isOpen, onClose, initialTab = "signup" }: AuthModalProps) {
  const router = useRouter();
  const [activeTab, setActiveTab] = useState<"signup" | "login">(initialTab);
  
  // Form State
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");

  // UI State: 'default' | 'submitting' | 'error' | 'success'
  const [status, setStatus] = useState<"default" | "submitting" | "error" | "success">("default");
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  const emailInputRef = useRef<HTMLInputElement>(null);

  // Sync initial tab when modal opens
  useEffect(() => {
    if (isOpen) {
      setActiveTab(initialTab);
      setStatus("default");
      setErrorMessage(null);
      setTimeout(() => emailInputRef.current?.focus(), 100);
    }
  }, [isOpen, initialTab]);

  // Keyboard 'Escape' listener for accessibility
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === "Escape" && isOpen && status !== "submitting") {
        onClose();
      }
    };
    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [isOpen, status, onClose]);

  if (!isOpen) return null;

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setStatus("submitting");
    setErrorMessage(null);

    try {
      if (activeTab === "signup") {
        // Step 1: Call POST /api/auth/signup
        const res = await fetch("/api/auth/signup", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ name, email, password }),
        });

        const data = await res.json();

        if (!res.ok) {
          setStatus("error");
          setErrorMessage(data.error || "Failed to create account. Please try again.");
          return;
        }

        // Step 2: Auto Sign-In after successful signup (REQ-03)
        const signInResult = await signIn("credentials", {
          email,
          password,
          redirect: false,
        });

        if (signInResult?.error) {
          setStatus("error");
          setErrorMessage("Account created, but automatic sign in failed. Please log in.");
          setActiveTab("login");
          return;
        }

        // Save prior login flag in localStorage for repeat visits
        localStorage.setItem("salesflow_has_account", "true");
        setStatus("success");
        onClose();
        router.push("/booking");
      } else {
        // Log In tab
        const signInResult = await signIn("credentials", {
          email,
          password,
          redirect: false,
        });

        if (signInResult?.error) {
          setStatus("error");
          setErrorMessage("Invalid email or password. Please check your credentials.");
          return;
        }

        localStorage.setItem("salesflow_has_account", "true");
        setStatus("success");
        onClose();
        router.push("/booking");
      }
    } catch (err) {
      console.error("Auth submit error:", err);
      setStatus("error");
      setErrorMessage("An unexpected network error occurred. Please check your connection.");
    }
  };

  return (
    <div
      className="fixed inset-0 z-50 bg-black/60 backdrop-blur-sm flex items-center justify-center p-0 sm:p-4 animate-in fade-in duration-200"
      onClick={(e) => {
        if (e.target === e.currentTarget && status !== "submitting") onClose();
      }}
    >
      {/* Modal Container: Full-screen on mobile (<=640px), centered rounded card on desktop */}
      <div className="bg-card text-card-foreground border border-border w-full h-full sm:h-auto sm:max-w-md sm:rounded-2xl shadow-2xl flex flex-col justify-between overflow-hidden relative animate-in zoom-in-95 duration-200">
        
        {/* Header with Close Button */}
        <div className="p-6 pb-0 flex items-center justify-between">
          <div className="flex border-b border-border w-full gap-6">
            <button
              type="button"
              onClick={() => {
                setActiveTab("signup");
                setErrorMessage(null);
              }}
              disabled={status === "submitting"}
              className={`pb-3 text-base font-semibold transition-colors relative cursor-pointer ${
                activeTab === "signup" ? "text-primary" : "text-muted-foreground hover:text-foreground"
              }`}
            >
              Sign Up
              {activeTab === "signup" && (
                <span className="absolute bottom-0 left-0 right-0 h-0.5 bg-primary rounded-full" />
              )}
            </button>
            <button
              type="button"
              onClick={() => {
                setActiveTab("login");
                setErrorMessage(null);
              }}
              disabled={status === "submitting"}
              className={`pb-3 text-base font-semibold transition-colors relative cursor-pointer ${
                activeTab === "login" ? "text-primary" : "text-muted-foreground hover:text-foreground"
              }`}
            >
              Log In
              {activeTab === "login" && (
                <span className="absolute bottom-0 left-0 right-0 h-0.5 bg-primary rounded-full" />
              )}
            </button>
          </div>

          <button
            type="button"
            onClick={onClose}
            disabled={status === "submitting"}
            className="absolute top-5 right-5 p-2 text-muted-foreground hover:text-foreground rounded-full hover:bg-muted/50 transition-colors cursor-pointer"
            aria-label="Close modal"
          >
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* Form Body */}
        <form onSubmit={handleSubmit} className="p-6 space-y-4 flex-1">
          {/* Name Field (Sign Up Only) */}
          {activeTab === "signup" && (
            <div className="space-y-1.5">
              <label htmlFor="name" className="text-xs font-medium font-mono text-muted-foreground uppercase tracking-wider block">
                Full Name <span className="text-muted-foreground/60">(Optional)</span>
              </label>
              <div className="relative">
                <User className="w-4 h-4 absolute left-3.5 top-3.5 text-muted-foreground" />
                <input
                  id="name"
                  type="text"
                  placeholder="Priya Sharma"
                  value={name}
                  onChange={(e) => setName(e.target.value)}
                  disabled={status === "submitting"}
                  className="w-full pl-10 pr-4 py-2.5 rounded-xl border border-border bg-input text-foreground text-sm focus:outline-none focus:ring-2 focus:ring-ring focus:border-transparent transition-all disabled:opacity-50"
                />
              </div>
            </div>
          )}

          {/* Email Field */}
          <div className="space-y-1.5">
            <label htmlFor="email" className="text-xs font-medium font-mono text-muted-foreground uppercase tracking-wider block">
              Email Address
            </label>
            <div className="relative">
              <Mail className="w-4 h-4 absolute left-3.5 top-3.5 text-muted-foreground" />
              <input
                id="email"
                ref={emailInputRef}
                type="email"
                required
                placeholder="name@company.com"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                disabled={status === "submitting"}
                className="w-full pl-10 pr-4 py-2.5 rounded-xl border border-border bg-input text-foreground text-sm focus:outline-none focus:ring-2 focus:ring-ring focus:border-transparent transition-all disabled:opacity-50"
              />
            </div>
          </div>

          {/* Password Field */}
          <div className="space-y-1.5">
            <label htmlFor="password" className="text-xs font-medium font-mono text-muted-foreground uppercase tracking-wider block">
              Password
            </label>
            <div className="relative">
              <Lock className="w-4 h-4 absolute left-3.5 top-3.5 text-muted-foreground" />
              <input
                id="password"
                type="password"
                required
                placeholder="••••••••"
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                disabled={status === "submitting"}
                className="w-full pl-10 pr-4 py-2.5 rounded-xl border border-border bg-input text-foreground text-sm focus:outline-none focus:ring-2 focus:ring-ring focus:border-transparent transition-all disabled:opacity-50"
              />
            </div>
          </div>

          {/* Reserved-Height Error Slot (Prevents layout jump per PRD Screen 02 wireframe) */}
          <div className="min-h-[44px] flex items-center">
            {status === "error" && errorMessage && (
              <div className="w-full p-2.5 rounded-xl bg-destructive/10 border border-destructive/20 text-destructive text-xs font-medium flex items-start gap-2 animate-in fade-in duration-150">
                <AlertCircle className="w-4 h-4 flex-shrink-0 mt-0.5" />
                <span>{errorMessage}</span>
              </div>
            )}
          </div>

          {/* Submit Button */}
          <button
            type="submit"
            disabled={status === "submitting"}
            className="w-full py-3 px-4 rounded-xl bg-primary text-primary-foreground font-semibold text-sm hover:bg-primary/90 transition-all shadow-sm hover:shadow flex items-center justify-center gap-2 disabled:opacity-60 cursor-pointer"
          >
            {status === "submitting" ? (
              <>
                <Loader2 className="w-4 h-4 animate-spin" />
                <span>{activeTab === "signup" ? "Creating account..." : "Logging in..."}</span>
              </>
            ) : (
              <span>{activeTab === "signup" ? "Create Account & Continue" : "Log In & Continue"}</span>
            )}
          </button>
        </form>

        {/* Footer Toggle Hint */}
        <div className="p-4 border-t border-border bg-muted/20 text-center text-xs text-muted-foreground">
          {activeTab === "signup" ? (
            <p>
              Already have an account?{" "}
              <button
                type="button"
                onClick={() => setActiveTab("login")}
                className="text-primary font-semibold hover:underline cursor-pointer"
              >
                Log In
              </button>
            </p>
          ) : (
            <p>
              Don&apos;t have an account yet?{" "}
              <button
                type="button"
                onClick={() => setActiveTab("signup")}
                className="text-primary font-semibold hover:underline cursor-pointer"
              >
                Sign Up
              </button>
            </p>
          )}
        </div>
      </div>
    </div>
  );
}
```
